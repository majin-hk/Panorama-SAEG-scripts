Option Explicit



'Table.SQLRequest = ""
Dim sqlClause
Dim currentDevice

' =========================================================
' 1. 获取当前设备名 (直接从界面控件读取)
' =========================================================
' 既然不能写死，我们就完全信任界面上 Filter_Device 的值。
' 注意：必须确保页面打开时，Filter_Device 已经被外部参数填入了正确的值。
currentDevice = Filter_Device.Value

' =========================================================
' 2. 清空其他查询条件 (修复报错版)
' =========================================================
' 既然系统不支持 IsObject，我们直接赋值，默认控件都存在。
' ---------------------------------------------------------

' 清空文本框（文本框可以为空）
Filter_JobOrder.Value = ""
Filter_Comment.Value = ""

' !!! 关于时间控件报错的修复 !!!
' 之前报错是因为时间控件不能等于 ""。
' 为了不报错，我们将清空时间的动作注释掉，或者你可以赋值为 Now (当前时间)。
' 这里选择注释掉，保留默认时间，保证脚本不崩溃。
' t_StartTime.Value = "" 
' t_EndTime.Value = ""

' =========================================================
' 3. 构建 SQL 语句 (去除不兼容函数)
' =========================================================
sqlClause = "WHERE 1=1 " 

' 既然系统不支持 Replace，我们直接拼接字符串。
' 只要设备名里不包含单引号(')等特殊字符，这样写是没问题的。
If currentDevice <> "" Then
	sqlClause = sqlClause & "AND [Device Number] = '" & currentDevice & "' "
End If

sqlClause = sqlClause & "ORDER BY [Start Time] DESC"

' =========================================================
' 4. 执行查询
' =========================================================
' 调试建议：如果发现查出来的数据不对（比如查出了上个设备的），
' 很可能是因为页面刚打开时，Filter_Device 的值还没来得及更新。
' 那属于组态软件的“加载顺序”问题，脚本逻辑本身没问题。
PnTrace "当前识别设备: " & currentDevice & " | SQL: " & sqlClause

Table.SQLRequest = sqlClause
Table.Read
