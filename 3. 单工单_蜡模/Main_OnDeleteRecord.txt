' =========================================================
' 【核心函数】通用日期解析器
' =========================================================
Function ParseUniversalDate(strInput)
	Dim parts, sDatePart, timePart, dateSubParts
	Dim y, m, d
    
	parts = Split(Trim(strInput), " ")
	If UBound(parts) >= 0 Then sDatePart = parts(0) Else ParseUniversalDate = "NULL" : Exit Function
	If UBound(parts) >= 1 Then timePart = parts(1) Else timePart = "00:00:00"
    
	If InStr(sDatePart, "/") > 0 Then
		dateSubParts = Split(sDatePart, "/")
	ElseIf InStr(sDatePart, "-") > 0 Then
		dateSubParts = Split(sDatePart, "-")
	ElseIf InStr(sDatePart, ".") > 0 Then
		dateSubParts = Split(sDatePart, ".")
	Else
		ParseUniversalDate = strInput : Exit Function
	End If
    
	If UBound(dateSubParts) = 2 Then
		If Len(dateSubParts(0)) = 4 Then
			y = dateSubParts(0): m = Right("0" & dateSubParts(1), 2): d = Right("0" & dateSubParts(2), 2)
		ElseIf Len(dateSubParts(2)) = 4 Then
			d = Right("0" & dateSubParts(0), 2): m = Right("0" & dateSubParts(1), 2): y = dateSubParts(2)
		Else
			ParseUniversalDate = strInput : Exit Function
		End If
		ParseUniversalDate = y & "-" & m & "-" & d & " " & timePart & ".0000000"
	Else
		ParseUniversalDate = strInput
	End If
End Function
