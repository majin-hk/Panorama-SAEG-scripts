Option Explicit

Dim oParent
Dim sNewJob
Dim sLastJob
Dim sCurrentUser
Dim sUserToSave
Dim oEditObject
Dim bHasPreviousJob
Dim dtNow

PnTrace ">>> [1] SaveJobOrderRecord 脚本开始"

Set oParent = PnGetObjectEx("#PARENT")

If (oParent Is Nothing) Then
	PnTrace ">>> [严重错误] 父对象未找到！"
	This.Active = False
End If

' 新工单号来自触发值
sNewJob = Trim(This.TriggeringValue & "")
sLastJob = Trim(oParent.Last_JobOrder & "")

PnTrace ">>> [2] TriggeringValue = [" & CStr(This.TriggeringValue) & "]"
PnTrace ">>> [3] 新工单 sNewJob = [" & sNewJob & "]"
PnTrace ">>> [4] 上一工单 Last_JobOrder = [" & sLastJob & "]"
PnTrace ">>> [5] 上一开始时间 Last_StartTime = [" & CStr(oParent.Last_StartTime) & "]"

If sNewJob = "" Or sNewJob = "0" Then
	PnTrace ">>> [停止点 A] 新工单为空或为0，不处理"
	This.Active = False
End If

dtNow = CDate(Now)

' 判断是否存在上一张有效工单
bHasPreviousJob = False
If sLastJob <> "" And sLastJob <> "0" Then
	If Trim(oParent.Last_StartTime & "") <> "" And Trim(oParent.Last_StartTime & "") <> "0" Then
		bHasPreviousJob = True
	End If
End If

' ============================================================
' 逻辑 A：先写上一工单
' ============================================================
If bHasPreviousJob = True Then

	PnTrace ">>> [6] 检测到上一工单，准备写数据库"

	' 用户：优先用当前界面上最终值；若为空再退回缓存
	sUserToSave = Trim(oParent.A5 & "")
	If sUserToSave = "" Then
		sUserToSave = Trim(oParent.Last_User & "")
	End If
	If sUserToSave = "" Then
		sUserToSave = "AUTO"
	End If

	' 写入数据库缓冲字段
	oParent.DB1 = sLastJob                  ' 上一工单号（缓存）
	oParent.DB2 = oParent.Last_StartTime    ' 上一工单开始时间（缓存）
	oParent.DB3 = dtNow                     ' 上一工单结束时间（当前）
	oParent.DB4 = oParent.A4                ' 当前设备号
	oParent.DB5 = sUserToSave               ' 当前最终操作员
	oParent.DB6 = Trim(oParent.A6 & "")     ' 当前最终备注
	oParent.DB7 = Trim(oParent.A7 & "")     ' 当前最终预热炉编号

	PnTrace ">>> [7] 写库快照：" & _
		" DB1=[" & CStr(oParent.DB1) & "]" & _
		" DB2=[" & CStr(oParent.DB2) & "]" & _
		" DB3=[" & CStr(oParent.DB3) & "]" & _
		" DB4=[" & CStr(oParent.DB4) & "]" & _
		" DB5=[" & CStr(oParent.DB5) & "]" & _
		" DB6=[" & CStr(oParent.DB6) & "]" & _
		" DB7=[" & CStr(oParent.DB7) & "]"

	Set oEditObject = PnGetObjectEx("Edit_JobOrder")

	If oEditObject Is Nothing Then
		PnTrace ">>> [严重错误] 找不到 Edit_JobOrder 对象！"
	Else
		PnTrace ">>> [8] 已找到 Edit_JobOrder，开始写库"

		oEditObject.ApplicationToDataBase = True
		PnTrace ">>> [9] ApplicationToDataBase = True"
		PnSleep 500

		oEditObject.DataBaseInsert = True
		PnTrace ">>> [10] DataBaseInsert = True"
		PnSleep 1000

		oEditObject.DataBaseInsert = False
		PnTrace ">>> [11] DataBaseInsert = False"
		PnSleep 300
	End If

Else
	PnTrace ">>> [6] 没有上一张有效工单，本次不写库，只登记开始时间"
End If

' ============================================================
' 逻辑 B：登记当前工单
' ============================================================
sCurrentUser = Trim(oParent.First_Input_User & "")
If sCurrentUser = "" Then
	sCurrentUser = Trim(oParent.A5 & "")
End If
If sCurrentUser = "" Then
	sCurrentUser = "AUTO"
End If

' 当前显示字段
oParent.A1 = sNewJob
oParent.A2 = dtNow
oParent.A5 = sCurrentUser

' 只缓存那些“在工单开始瞬间就确定”的字段
oParent.Last_JobOrder = sNewJob
oParent.Last_StartTime = dtNow
oParent.Last_User = sCurrentUser

PnTrace ">>> [12] 当前工单登记完成：" & _
	" A1=[" & CStr(oParent.A1) & "]" & _
	" A2=[" & CStr(oParent.A2) & "]" & _
	" A4=[" & CStr(oParent.A4) & "]" & _
	" A5=[" & CStr(oParent.A5) & "]" & _
	" A6=[" & CStr(oParent.A6) & "]" & _
	" A7=[" & CStr(oParent.A7) & "]"

PnTrace ">>> [13] Last_* 缓存完成：" & _
	" Last_JobOrder=[" & CStr(oParent.Last_JobOrder) & "]" & _
	" Last_StartTime=[" & CStr(oParent.Last_StartTime) & "]" & _
	" Last_User=[" & CStr(oParent.Last_User) & "]"

PnTrace ">>> [14] SaveJobOrderRecord 脚本结束"
