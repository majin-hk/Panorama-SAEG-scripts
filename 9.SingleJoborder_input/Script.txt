Option Explicit

Dim myApp
Dim scanText

Set myApp = PnGetObjectEx("#PARENT")

If myApp Is Nothing Then
	PnTrace ">>> [严重错误] #PARENT 未找到"
	This.Active = False
End If

PnTrace ">>> [触发] ScanValidateFlag=" & CStr(myApp.ScanValidateFlag) & _
	" | TriggeringValue=" & CStr(This.TriggeringValue) & _
	" | ScanInput=[" & CStr(myApp.ScanInput) & "]"

If myApp.ScanValidateFlag <> True Then
	PnTrace ">>> [忽略] 非 True 触发"
	This.Active = False
End If

scanText = Trim(CStr(myApp.ScanInput & ""))

If scanText <> "" Then
	myApp.TargetJobOrder = scanText
	PnTrace ">>> [成功] TargetJobOrder=[" & CStr(myApp.TargetJobOrder) & "]"
Else
	PnTrace ">>> [警告] ScanInput 为空"
End If

myApp.ScanInput = ""
myApp.ScanValidateFlag = False
