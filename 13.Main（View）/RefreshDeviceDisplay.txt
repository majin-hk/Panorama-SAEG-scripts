var APP = PnGetObjectEx("/$_SYSTEM/APP") as CODRA_System.AppConfig;

if (APP == null)
{
	PnTrace("RefreshDeviceDisplay失败：找不到当前会话 /$_SYSTEM/APP", FuncScript.TRACE_LEVEL.ERROR);
	return;
}

string currentLanguage = APP.CurrentLanguage.ToString();
int count = 0;

PnTrace("RefreshDeviceDisplay开始执行，CurrentLanguage=" + currentLanguage, FuncScript.TRACE_LEVEL.INFO);

// ==================== 1. Device_Wax 类设备 ====================
string[] deviceWaxPaths = new string[]
{
	"/Main/Wax/MPI_1/Device",
	"/Main/Wax/MPI_2/Device",
	"/Main/Wax/MPI_3/Device",
	"/Main/Wax/MPI_4/Device",
	"/Main/Wax/CTM_1/Device",
	"/Main/Wax/CTM_2/Device",
	"/Main/Wax/CTM_3/Device",
	"/Main/Sand_blasting/Sand_blasting1/Device",
	"/Main/Sand_blasting/Sand_blasting2/Device",
	"/Main/Sand_blasting/Sand_blasting3/Device",
	"/Main/Sand_blasting/Sand_blasting4/Device",
	"/Main/Sand_blasting/Sand_blasting5/Device"
};

foreach (var path in deviceWaxPaths)
{
	var dev = PnGetObjectEx(path) as Device_SCADA.Device_Wax;
	if (dev == null)
	{
		PnTrace("RefreshDeviceDisplay警告：无法转换为 Device_SCADA.Device_Wax -> " + path, FuncScript.TRACE_LEVEL.WARNING);
		continue;
	}

	if (currentLanguage == "4")
	{
		dev.Display_Device_Name = dev.Device_Name_zh;
		dev.Display_Device_Type = dev.Device_Type_zh;
	}
	else if (currentLanguage == "9")
	{
		dev.Display_Device_Name = dev.Device_Name_en;
		dev.Display_Device_Type = dev.Device_Type_en;
	}
	else
	{
		dev.Display_Device_Name = dev.Device_Name_en;
		dev.Display_Device_Type = dev.Device_Type_en;
	}

	count++;
}
