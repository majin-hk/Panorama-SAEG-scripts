Option Explicit

Dim sJobOrder, sBatchID, sComment
Dim oRecord, oEdit
Dim sPath

' 1. 启动调试
PnTrace ">>> [删除脚本] 正在解析选中行数据..."

' 2. 外部校验：判断是否有选中行
If Table.Value <> "" Then
	' Table.Value 通常是以 "|" 分隔的字符串
	oRecord = Split(Table.Value, "|")

	' 【检查列数】确保数组长度足够
	If UBound(oRecord) >= 3 Then
        
		' 【路径检查】指向您的 Edit_JobOrder 对象
		sPath = "../#PARENT/Edit_JobOrder"
		Set oEdit = PnGetObjectEx(sPath)

		If oEdit Is Nothing Then
			PnTrace ">>> [错误] 找不到 Edit_JobOrder 数据库对象！"
		Else
			' =======================================================
			' 【步骤 A】 获取数据 (严格对应截图列顺序)
			' =======================================================
            
			' Column 0: Job Order (工单号)
			sJobOrder = Trim(oRecord(0))
            
			' Column 1: Batch Number (批次号)
			sBatchID = Trim(oRecord(1))
            
			' Column 6: Comments (备注) - 假设在第7列
			If UBound(oRecord) >= 6 Then 
				sComment = Trim(oRecord(6))
			Else 
				sComment = ""
			End If

			PnTrace ">>> [选中行] 工单: " & sJobOrder & " | 批次: " & sBatchID

			' =======================================================
			' 【步骤 B】 构造筛选条件 (联合定位，防止误删)
			' =======================================================

			' --- 条件 1: Job Order (必须精确匹配，防止删掉同批次其他工单) ---
			oEdit.Field1 = "Job Order"
			oEdit.Test1 = DB_SQL_TEST.EQ
			oEdit.Value1 = sJobOrder
			oEdit.Operator1 = DB_LOGICAL_OPERATOR.And

			' --- 条件 2: Batch Number ---
			oEdit.Field2 = "Batch Number"
			oEdit.Test2 = DB_SQL_TEST.EQ
			oEdit.Value2 = sBatchID
			oEdit.Operator2 = DB_LOGICAL_OPERATOR.And

			' --- 条件 3: Start Time (索引 2) ---
			If Trim(oRecord(2) & "") <> "" Then
				oEdit.Field3 = "Start Time"
				oEdit.Test3 = DB_SQL_TEST.EQ
				oEdit.Value3 = ParseUniversalDate(oRecord(2))
			Else
				oEdit.Field3 = "Start Time"
				oEdit.Test3 = DB_SQL_TEST.IS_NULL
			End If
			oEdit.Operator3 = DB_LOGICAL_OPERATOR.And

			' --- 条件 4: End Time (索引 3) ---
			If Trim(oRecord(3) & "") <> "" Then
				oEdit.Field4 = "End Time"
				oEdit.Test4 = DB_SQL_TEST.EQ
				oEdit.Value4 = ParseUniversalDate(oRecord(3))
			Else
				oEdit.Field4 = "End Time"
				oEdit.Test4 = DB_SQL_TEST.IS_NULL
			End If

			' =======================================================
			' 【步骤 C】 执行删除
			' =======================================================
			PnTrace ">>> [执行] 正在从数据库删除记录..."
			oEdit.DataBaseDelete = True

			' 等待并刷新前端表格
			PnSleep 300
			Table.Read 
			PnTrace ">>> [完成] 记录已删除并刷新列表。"
		End If
        
	Else
		PnTrace ">>> [错误] 选中行数据列数不足，无法解析。"
	End If
Else
	PnTrace ">>> [警告] 请先在表格中选中一行记录。"
End If


' =========================================================
' 【核心函数】日期解析器 (将表格日期转为数据库标准格式)
' =========================================================
Function ParseUniversalDate(strInput)
	Dim parts, sDatePart, timePart, dateSubParts
	Dim y, m, d
    
	strInput = Trim(strInput)
	If strInput = "" Then ParseUniversalDate = "" : Exit Function

	parts = Split(strInput, " ")
	sDatePart = parts(0)
	If UBound(parts) >= 1 Then timePart = parts(1) Else timePart = "00:00:00"
    
	' 处理 / 或 - 或 . 分隔符
	If InStr(sDatePart, "/") > 0 Then
		dateSubParts = Split(sDatePart, "/")
	ElseIf InStr(sDatePart, "-") > 0 Then
		dateSubParts = Split(sDatePart, "-")
	Else
		ParseUniversalDate = strInput : Exit Function
	End If
    
	If UBound(dateSubParts) = 2 Then
		' 判断是 YYYY/MM/DD 还是 DD/MM/YYYY (根据截图是 DD/MM/YYYY)
		If Len(dateSubParts(0)) = 4 Then
			y = dateSubParts(0): m = dateSubParts(1): d = dateSubParts(2)
		Else
			d = dateSubParts(0): m = dateSubParts(1): y = dateSubParts(2)
		End If
		ParseUniversalDate = y & "-" & Right("0" & m, 2) & "-" & Right("0" & d, 2) & " " & timePart
	Else
		ParseUniversalDate = strInput
	End If
End Function
