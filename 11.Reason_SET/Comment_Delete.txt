string error;
string selected = Tabel.Value;

PnTrace("当前选中值 = " + selected, FuncScript.TRACE_LEVEL.INFO);

if (selected == null || selected.Trim() == "")
{
	PnTrace("删除失败：未选中任何行。", FuncScript.TRACE_LEVEL.WARNING);
	return;
}

// 当前顺序：ID|CommentText
string[] fields = selected.Split('|');

if (fields.Length < 1)
{
	PnTrace("删除失败：无法解析选中行。", FuncScript.TRACE_LEVEL.ERROR);
	return;
}

string id = fields[0].Trim();

if (id == "")
{
	PnTrace("删除失败：选中行没有 ID。", FuncScript.TRACE_LEVEL.ERROR);
	return;
}

var Edit_DB = PnGetObjectEx("../Edit_DB") as CODRA_DataBase.DBTable;

if (Edit_DB == null)
{
	PnTrace("删除失败：无法获取 Edit_DB 对象。", FuncScript.TRACE_LEVEL.ERROR);
	return;
}

// 清理旧条件
Edit_DB.UseFilterAndOrderClause = false;
Edit_DB.FilterAndOrderClause = "";

// 配置删除条件：WHERE ID = ...
Edit_DB.Field1 = "ID";
Edit_DB.Test1 = CODRA_DataBase.DB_SQL_TEST.EQ;
Edit_DB.Value1 = id;

// 执行删除
var res = Edit_DB.Delete(out error);

if (error != null && error != "")
{
	PnTrace("删除失败：" + error, FuncScript.TRACE_LEVEL.ERROR);
	return;
}

PnTrace("删除成功，ID=" + id, FuncScript.TRACE_LEVEL.INFO);

// 刷新 Viewer
Tabel.Read();
Tabel.UnSelect();

var Reason_SET = (PnGetObjectEx("../#PARENT") as Device_SCADA.Reason_SET);
Comment_1.InputList = Reason_SET.CommentStore;
Comment_1.Value = "";

string rebuildError;
string sep = Comment_1.Separator;

if (sep == null || sep == "")
{
	sep = ";";
	Comment_1.Separator = sep;
}

var rebuildDB = PnGetObjectEx("../Edit_DB") as CODRA_DataBase.DBTable;
var colComment = PnGetObjectEx("../Edit_DB/A2") as CODRA_DataBase.DBTableColumn;

if (rebuildDB == null || colComment == null)
{
	PnTrace("重建下拉框失败：无法获取数据库对象。", FuncScript.TRACE_LEVEL.ERROR);
	return;
}

// 清理旧条件
rebuildDB.UseFilterAndOrderClause = true;
rebuildDB.FilterAndOrderClause = "ORDER BY ID ASC";
rebuildDB.MaxRecords = 0;
rebuildDB.Field1 = "";
rebuildDB.Value1 = "";
rebuildDB.Field2 = "";
rebuildDB.Value2 = "";
rebuildDB.Field3 = "";
rebuildDB.Value3 = "";
rebuildDB.Field4 = "";
rebuildDB.Value4 = "";

// 先读取整表
rebuildDB.Read(out rebuildError);

if (rebuildError != "")
{
	PnTrace("重建下拉框失败：" + rebuildError, FuncScript.TRACE_LEVEL.ERROR);
	return;
}

string inputList = "";

if (rebuildDB.RecordCount > 0)
{
	// 先处理当前记录
	if (colComment.DataBaseValue != null)
	{
		string commentText = colComment.DataBaseValue.ToString().Trim();

		if (commentText != "")
			inputList = commentText;
	}

	// 再逐条读后续记录
	while (rebuildDB.GoToNextRecord())
	{
		if (colComment.DataBaseValue == null)
			continue;

		string commentText = colComment.DataBaseValue.ToString().Trim();

		if (commentText == "")
			continue;

		if (inputList == "")
			inputList = commentText;
		else
			inputList = inputList + sep + commentText;
	}
}

// 清理状态
rebuildDB.UseFilterAndOrderClause = false;
rebuildDB.FilterAndOrderClause = "";
rebuildDB.MaxRecords = 0;

// 同步到缓存和下拉框
Reason_SET.CommentStore = inputList;
Comment_1.InputList = Reason_SET.CommentStore;
Comment_1.Value = "";

PnTrace("Reason_SET.CommentStore = " + Reason_SET.CommentStore, FuncScript.TRACE_LEVEL.INFO);
PnTrace("Comment_1.InputList = " + Comment_1.InputList, FuncScript.TRACE_LEVEL.INFO);
PnTrace("下拉框重建完成。", FuncScript.TRACE_LEVEL.INFO);
