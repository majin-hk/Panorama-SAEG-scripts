Option Explicit

Dim Start
Dim oParent
Dim sRecordUser
Dim oEditObject

' 1. 获取父对象引用
Set oParent = PnGetObjectEx("#PARENT")

If (oParent Is Nothing) Then
	PnTrace ">>> [严重错误] 父对象未找到！"
	This.Active = False
End If

' 2. 获取状态
Start = PnGetValue("#PARENT.Working")

PnTrace "--------------------------------------------------"
PnTrace ">>> [脚本触发] 状态变化为: " & CStr(Start)

' ============================================================
' 逻辑 A：启动瞬间 (Start 变为 True)
' ============================================================
If Start = True Then

	' 记录开始时间
	oParent.A2 = CDate(Now)

	' 固化本次单工单用户：只认首次输入用户
	sRecordUser = oParent.First_Input_User
	If Trim(sRecordUser & "") = "" Then
		sRecordUser = "AUTO"
	End If

	' 假设 A5 为操作员字段（如不是，请替换为你的真实字段）
	oParent.A5 = sRecordUser

	PnTrace ">>> [启动逻辑] A2 (开始时间) 已设置为: " & CStr(oParent.A2)
	PnTrace ">>> [启动逻辑] 首次输入用户已固化为: " & CStr(sRecordUser)
	PnTrace ">>> [提示] 此时不读取 A1/A6，等待生产结束..."

End If

' ============================================================
' 逻辑 B：停止瞬间 (Start 变为 False) —— 保存数据
' ============================================================
If Start = False Then

	PnTrace ">>> [停止逻辑] 准备保存数据..."

	' 1. 记录结束时间
	oParent.A3 = CDate(Now)
	PnTrace ">>> [步骤1] A3 (结束时间): " & CStr(oParent.A3)

	' 2. 获取本次固定用户
	sRecordUser = oParent.First_Input_User
	If Trim(sRecordUser & "") = "" Then
		sRecordUser = oParent.A5
	End If
	If Trim(sRecordUser & "") = "" Then
		sRecordUser = "AUTO"
	End If

	' 每次写库前都重新固定用户字段，避免异步读取串值
	oParent.A5 = sRecordUser

	' 3. 打印关键值
	PnTrace ">>> [步骤2] 获取输入值 -> 工单(A1):[" & CStr(oParent.A1) & "] | 产品(A6):[" & CStr(oParent.A6) & "]"
	PnTrace ">>> [步骤2] 检查开始时间 -> A2:[" & CStr(oParent.A2) & "]"
	PnTrace ">>> [步骤2] 本次记录用户 -> A5:[" & CStr(oParent.A5) & "] | First_Input_User:[" & CStr(oParent.First_Input_User) & "]"

	' ============================================================
	' 判断 A1 是否为空
	' ============================================================
	If Trim(oParent.A1 & "") = "" Then
		PnTrace ">>> [拦截] 警告：工单(A1)为空或界面上无值，已取消写入数据库！"
	Else
		PnTrace ">>> [步骤3] 字段赋值完成，准备执行 Insert..."

		Set oEditObject = PnGetObjectEx("Edit_JobOrder")

		If oEditObject Is Nothing Then
			PnTrace ">>> [严重错误] 找不到 Edit_JobOrder 对象！路径可能不对。"
		Else
			PnTrace ">>> [执行] 找到对象，正在插入数据库..."
			PnTrace ">>> [写库快照] A1=" & CStr(oParent.A1) & _
				", A2=" & CStr(oParent.A2) & _
				", A3=" & CStr(oParent.A3) & _
				", A5=" & CStr(oParent.A5) & _
				", A6=" & CStr(oParent.A6)

			oEditObject.ApplicationToDataBase = True
			PnSleep 1000
			oEditObject.DataBaseInsert = True
			PnTrace ">>> [成功] DataBaseInsert 指令已发送。"

			' 补充复位，避免残留状态
			PnSleep 200
			oEditObject.DataBaseInsert = False
		End If

		PnTrace ">>> [步骤4] DataBaseInsert 指令已发送 (成功)"

		' 写入完成后重置界面 / 当前工单信息
		oParent.A1 = ""
		oParent.A6 = ""
		oParent.A5 = ""
		oParent.First_Input_User = ""

		PnSleep 300
		PnTrace ">>> [结束] 界面已重置"
	End If
End If
