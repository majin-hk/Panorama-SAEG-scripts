' =========================================================
' 【核心函数】日期解析器 (适配 DD/MM/YYYY)
' =========================================================
Function ParseUniversalDate(strInput)
	Dim parts, sDatePart, timePart, dateSubParts
	Dim y, m, d
    
	strInput = Trim(strInput)
	If strInput = "" Then ParseUniversalDate = "" : Exit Function

	parts = Split(strInput, " ")
	sDatePart = parts(0)
	If UBound(parts) >= 1 Then timePart = parts(1) Else timePart = "00:00:00"
    
	' 处理多种分隔符
	If InStr(sDatePart, "/") > 0 Then
		dateSubParts = Split(sDatePart, "/")
	ElseIf InStr(sDatePart, "-") > 0 Then
		dateSubParts = Split(sDatePart, "-")
	Else
		ParseUniversalDate = strInput : Exit Function
	End If
    
	If UBound(dateSubParts) = 2 Then
		' 适配 20/03/2026 这种 DD/MM/YYYY 格式
		If Len(dateSubParts(0)) = 4 Then ' YYYY/MM/DD
			y = dateSubParts(0): m = dateSubParts(1): d = dateSubParts(2)
		Else ' DD/MM/YYYY
			d = dateSubParts(0): m = dateSubParts(1): y = dateSubParts(2)
		End If
		ParseUniversalDate = y & "-" & Right("0" & m, 2) & "-" & Right("0" & d, 2) & " " & timePart
	Else
		ParseUniversalDate = strInput
	End If
End Function
